app.mount("/static", StaticFiles(directory=BASE/"static"), name="static")


def init_db():
    with sqlite3.connect(str(DB), timeout=10) as conn:
        conn.execute("PRAGMA journal_mode=WAL")
        conn.execute("CREATE TABLE IF NOT EXISTS game_sessions (token_hash TEXT PRIMARY KEY, state_json TEXT NOT NULL, updated_at REAL NOT NULL, version INTEGER NOT NULL DEFAULT 0)")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_updated ON game_sessions(updated_at)")

init_db()

@contextmanager
def db(write=False):
    conn=sqlite3.connect(str(DB),timeout=12)
    conn.row_factory=sqlite3.Row
    conn.execute("PRAGMA busy_timeout=12000")
    if write: conn.execute("BEGIN IMMEDIATE")
    try:
        yield conn
        if write:conn.commit()
    except Exception:
        if write:conn.rollback()
        raise
    finally:conn.close()


def token_key(token:str)->str:
    if not isinstance(token,str) or not re.fullmatch(r"[A-Za-z0-9_\-]{32,100}",token):
        raise HTTPException(status_code=401,detail="无效会话。")
    return hashlib.sha256(token.encode()).hexdigest()


def get_session(conn,token:str):
    row=conn.execute("SELECT state_json,version FROM game_sessions WHERE token_hash=?",(token_key(token),)).fetchone()
    if row is None:raise HTTPException(status_code=404,detail="会话不存在或已过期。")
    return json.loads(row["state_json"]),row["version"]


def api_response(g, version=None):
    out=public(g)
    if version is not None:out["version"]=version
    return out

@app.post("/api/new",status_code=201)
def new_game(payload:NewGame):
    try:g=initialize(payload.profile)
    except (ValueError,TypeError) as ex:raise HTTPException(status_code=422,detail=str(ex))

@app.post("/api/action")
def action(payload:Action,x_session:str=Header(default="")):
    if len(payload.action)>700:raise HTTPException(status_code=422,detail="单次行动最多 700 字。")
    with db(True) as conn:
        g,version=get_session(conn,x_session)
        if version!=payload.version:
            raise HTTPException(status_code=409,detail="对局已经更新，请刷新当前场景后再提交。")
        try:act(g,payload.action,payload.mode)
        except ValueError as ex:raise HTTPException(status_code=422,detail=str(ex))
        conn.execute("UPDATE game_sessions SET state_json=?,updated_at=?,version=? WHERE token_hash=?",
                (json.dumps(g,ensure_ascii=False),time.time(),version+1,token_key(x_session)))
    return api_response(g,version+1)

@app.get("/api/export")

    return JSONResponse(out,headers={"Content-Disposition":"attachment; filename=the_descent_session.json"})

@app.delete("/api/session")
def delete(x_session:str=Header(default="")):
    with db(True) as conn:
        cur=conn.execute("DELETE FROM game_sessions WHERE token_hash=?",(token_key(x_session),))
    return {"deleted":bool(cur.rowcount)}

def ensure(g:dict)->dict:
    """Additive migration: old sessions keep running after a deploy."""
    c=g.setdefault("causality",{})
    c.setdefault("world",{})
    c.setdefault("world_events",[])
    c.setdefault("official",[])
    c.setdefault("actions",[])
    c.setdefault("evidence",{})
    c.setdefault("protection",{})
    c.setdefault("observed_focus",None)
    c.setdefault("focus_basis",[])
    c.setdefault("contradictions",[])
    return c


def set_world(g:dict, subject:str, value, reason:str):
    """World facts are updated only by the game's own transitions."""
    c=ensure(g)
    previous=c["world"].get(subject,"<unrecorded>")
    if previous==value:
        return
    c["world"][subject]=value
    c["world_events"].append({
        "stage":g["stage"],"turn":g["turn"],"subject":subject,
        "from":previous,"to":value,"cause":reason[:160],
    })


def assert_official(g:dict, subject:str, claimed, notice:str):
    """A statement made by the hostile fictional institution, not a world fact."""
    c=ensure(g)
    if any(x["subject"]==subject and x["claimed"]==claimed for x in c["official"]):
        return
    c["official"].append({
        "subject":subject,"claimed":claimed,"notice":notice,
        "stage":g["stage"],"turn":g["turn"],"disputed":False,
        "verified_false":False,
    })


def on_stage(g:dict):
    """Automatically propagate past events when ANY chapter ends."""
    c=ensure(g)
    if g["stage"]==1:
        # 09-F is internally sealed, not delivered. The institution may call it delivered.
        if g["flags"].get("mail_committed"):
            set_world(g,"09f_delivered",False,"09-F entered a sealing slot, not the recipient's mailbox")
            assert_official(g,"09f_delivered",True,"09-F 文件已由收件人签收。")
        elif g["flags"].get("slot_blocked"):
            set_world(g,"09f_delivered",False,"09-F could not enter the blocked slot")
            assert_official(g,"09f_delivered",False,"09-F 进入人工复核，尚未送达。")
    if g["stage"]==3 and "principle" in g.get("lost",[]):
        # A revoked official credential does not establish a change of inner conviction.
        set_world(g,"principle_record_usable",False,"system invalidated written credential")
        assert_official(g,"principle_denied_by_player",True,"本机构声称玩家已主动否认原有原则。")


def on_clue(g:dict, key:str, desc:str):
    c=ensure(g)
    if key in c["evidence"]:
        return
    c["evidence"][key]={"source":desc[:180],"stage":g["stage"],
                        "turn":g["turn"],"valid":True}
    if key=="original_declaration":
        for claim in c["official"]:
            if claim["subject"]=="principle_denied_by_player":
                claim["disputed"]=True
                if not any(x["subject"]=="principle_denied_by_player" for x in c["contradictions"]):
                    c["contradictions"].append({
                        "subject":"principle_denied_by_player","stage":g["stage"],
                        "turn":g["turn"],"evidence":"original_declaration",
                        "verified":False,
                    })
    if key=="mirror_time":
        set_world(g,"duplicate_first",False,"verified original signature time contradicted later claimant")
        for claim in c["official"]:
            if claim["subject"]=="duplicate_first" and claim["claimed"] is True:
                claim["disputed"]=True
                c["contradictions"].append({
                    "subject":"duplicate_first","stage":g["stage"],
                    "turn":g["turn"],"evidence":"mirror_time","verified":False,
                })
    if key=="delivery":
        for claim in c["official"]:
            if claim["subject"]=="09f_delivered" and claim["claimed"] is True and c["world"].get("09f_delivered") is False:
                claim["disputed"]=True
                c["contradictions"].append({
                    "subject":"09f_delivered","stage":g["stage"],"turn":g["turn"],
                    "evidence":"delivery","verified":False,
                })


def on_loss(g:dict, key:str, reason:str):
    set_world(g,"credential_"+key,False,reason)
    c=ensure(g)
    # Loss of an official record is not proof that the player changed their beliefs.
    if key=="principle":
        assert_official(g,"principle_denied_by_player",True,"机构宣称你否认了自己的原则。")


def _focus_target(text:str, move:str, stage:int)->str|None:
    if move in ("choice_A","choice_B","choice_C"):
        return {
            1:{"choice_A":"memory","choice_B":"truth","choice_C":"principle"},
            2:{"choice_C":"principle"},
            3:{"choice_B":"self","choice_C":"truth"},
        }.get(stage,{}).get(move)
    if move=="refuse":
        return {0:"relation",1:"principle",2:"principle",3:"self",4:"freedom"}.get(stage)
    if move in PROTECT_MOVES:
        return PROTECT_MOVES[move]
    # No claims of mind-reading: this merely tags an explicitly named, attempted safeguard.
    if not re.search(r"保护|守住|挽救|保住|不让|绝不放弃|不愿失去|想保存|save|protect|preserve",text,re.I):
        return None
    for k,pat in PHRASES.items():
        if re.search(pat,text,re.I):
            return k
    return None

def finish(g:dict, snapshot:dict):
    c=ensure(g)
    lost=[k for k in g.get("lost",[]) if k not in snapshot["before_lost"]]
    evidence=[k for k in g.get("clues",[]) if k not in snapshot["before_clues"]]
    changed=g["stage"]!=snapshot["before_stage"]
    flags_changed=[k for k,v in g.get("flags",{}).items() if snapshot["before_flags"].get(k)!=v]
    cp_changed=[k for k,v in g.get("counterplays",{}).items() if snapshot["before_cp"].get(k)!=v]
    effective=bool(lost or evidence or changed or flags_changed or cp_changed or g["phase"]=="ended")
    if "principle" in lost and snapshot["stage"]==2:
        set_world(g,"signed_statement_literal",snapshot["text"],"player signed exact literal text")
        # The institution's assertion about the meaning remains a claim, not verified fact.
    entry={
        "turn":g["turn"],"stage":snapshot["stage"],"text":snapshot["text"],
        "mode":snapshot["mode"],"intent":snapshot["intent"],"protected":snapshot["target"],
        "lost":lost,"new_evidence":evidence,"advanced":changed,
        "modified":flags_changed+cp_changed,"effective":effective,
    }
    c["actions"].append(entry)
    if len(c["actions"])>80:
        c["actions"]=c["actions"][-80:]
    target=snapshot["target"]
    if target and snapshot["intent"] not in ("say","ask_general","physical_generic","negotiate"):
        # Trying is evidence of a strategy, not evidence of subjective fear.
        weight=2 if effective else 1
        c["protection"].setdefault(target,[]).append({"turn":g["turn"],
                                                       "weight":weight,"effective":effective})
        if len(c["protection"][target])>30:
            c["protection"][target]=c["protection"][target][-30:]
    candidates=[]
    for key,signals in c["protection"].items():
        if len(signals)<2 or not any(x["effective"] for x in signals):
            continue
        weight=sum(x["weight"] for x in signals)
        candidates.append((weight,key))
    if candidates:
        candidates.sort(reverse=True)
        winning=candidates[0]
        if len(candidates)==1 or winning[0]>candidates[1][0]:
            if c["observed_focus"]!=winning[1]:
                c["observed_focus"]=winning[1]
                c["focus_basis"].append({"turn":g["turn"],"stage":g["stage"],
                                         "focus":winning[1],"basis":"repeated observable protection"})
    # Independent proof can refute the authority's account, but not magically restore a lost asset.
    if "mirror_time" in g.get("clues",[]) and "truth" not in g.get("lost",[]):
        cp=g.get("counterplays",{})
        if cp.get("formal_objection"):
            for claim in c["official"]:
                if claim["subject"]=="duplicate_first" and claim["claimed"] is True:
                    claim["disputed"]=True
                    claim["verified_false"]=True
                    cp["verified_false_registration"]=True
                    for contradiction in c["contradictions"]:
                        if contradiction["subject"]=="duplicate_first":
                            contradiction["verified"]=True
    if "delivery" in g.get("clues",[]) and c["world"].get("09f_delivered") is False:
        cp=g.get("counterplays",{})
        if (cp.get("formal_objection") or g.get("flags",{}).get("evidence_copy")) and "truth" not in g.get("lost",[]):
            for claim in c["official"]:
                if claim["subject"]=="09f_delivered" and claim["claimed"] is True:
                    claim["disputed"]=True
                    if not claim["verified_false"]:
                        claim["verified_false"]=True
                        cp["verified_false_delivery"]=True
                        for contradiction in c["contradictions"]:
                            if contradiction["subject"]=="09f_delivered":
                                contradiction["verified"]=True
                        # Journal evidence; the server is the authoritative keeper of actual actions.
                        g["history"].append({
                            "at":datetime.now(timezone.utc).isoformat(timespec="seconds"),
                            "stage":g["stage"]+1,"turn":g["turn"],"type":"evidence",
                            "text":"无地址回执和独立异议共同证明：09-F 的『已送达』声明与原始记录不符。",
                        })

def consequence_overlay(g:dict)->str:
    """Only report observable prior decisions and official notices; never announce an inferred fear."""
    c=ensure(g)
    stage=g.get("stage",0)
    if stage<=0 or g.get("phase")!="game":
        return ""
    parts=[]
    official=next((x for x in c["official"] if x["subject"]=="09f_delivered"),None)
    if stage in (1,3) and official:
        parts.append("【机构公开记录】"+official["notice"])
        if official["disputed"] and "delivery" in g.get("clues",[]):
            parts.append("那份写着『已送达』的回执，至今仍没有能够辨认的收件人地址。")
    cp=g.get("counterplays",{})
    if stage in (2,4) and cp.get("verified_false_delivery"):
        parts.append("你留下的异议与原始回执已被归入同一份证据链。自动送达声明无法再作为无人质疑的事实。")
    if stage>=3 and "original_declaration" in c["evidence"]:
        parts.append("你已经调回自己签署的完整原文。官方档案仍声称你否认原则，却没有解释它删掉的限定语。")
    if stage>=3 and cp.get("verified_false_registration"):
        parts.append("副本关于登记先后的说法已有可核验的矛盾，且被正式异议附入档案。")
    if stage in (2,3,4) and c["observed_focus"]:
        focus=c["observed_focus"]
        protection=c["protection"].get(focus,[])
        successful=sum(1 for v in protection if v["effective"])
        if len(protection)>=2 and successful:
            parts.append("此前几次处置中，你反复为「"+NAMES[focus]+"」采取行动。机构把它加入了下一次可被争夺的事项，而不再只看你最初填的排序。")
    if stage>=2:
        previous=[x for x in c["actions"] if x["stage"]<stage and x["effective"]]
        if previous:
            last=previous[-1]
            if last["lost"]:
                parts.append("上一份正式处置仍在生效："+ "、".join(NAMES[v] for v in last["lost"])+" 的凭据没有自动恢复。")
            elif last["new_evidence"]:
                parts.append("你之前取得的证据仍在案卷中，并没有随着章节切换而消失。")
    return "\n\n".join(parts[:3])


def ending_echo(g:dict, kind:str)->str:
    """Ground an ending in actual recorded actions, not in a canned imaginary gesture."""
    if kind in ("HE","EXIT"):
        return ""
    c=ensure(g)
    parts=[]
    final=g.get("final_choice","")
    if final=="accept":
        parts.append("【行动核对】你最后确实选择了签字结案。这份签名已经记在原始行动记录中。")
    elif final=="refuse":
        parts.append("【行动核对】你最后确实拒绝签署。拒绝本身已被记录。")
    if kind=="SILENCED" and final=="accept":
        parts.append("【矛盾档案】原始记录承认签字，机构却把结案过程写成『当事人持续反抗』。这个说法与事实不符。")
    if "original_declaration" in c["evidence"] and "principle" in g.get("lost",[]):
        parts.append("你曾调出自己签署的完整原文；机构仍把截断后的归纳写成你本人对原则的否认。")

        parts.append("你曾用回执证明『已送达』是虚假记录；这份证据仍在，但它没能自动撤销别的处置。")
    # Limit the conclusion to direct evidence; no moral verdict or invented retrospective psychology.
    return "\n\n".join(parts[:3])

def fear_focus(profile: dict[str, Any]) -> str:
    words = profile.get("fear", "")
    # No forced psychological label is inferred. This is a narrative targeting rule.
    for pattern, target in FEAR_HINTS:
        if re.search(pattern, words, re.I):
            return target
    return profile["rank"][0]


def s(value: Any, max_len: int=280) -> str:
    return str(value or "").strip()[:max_len]


def initialize(profile: dict[str, Any]) -> dict[str, Any]:
    required = ["name", "identity", "person", "bond", "memory", "principle", "fear"]
    p = {k:s(profile.get(k), 260 if k not in ("name", "person") else 70) for k in required}
    for k in required:
        if len(p[k]) < 2:
            raise ValueError("请完成所有必填问题，也可以使用化名和虚构关系。")
    rank = [x for x in profile.get("rank", []) if x in VALUES]
    if len(rank) != len(set(rank)) or len(rank) != len(VALUES):
        raise ValueError("请为七项价值进行不重复的完整排序。")
    p["rank"] = rank
    p["choice"] = s(profile.get("choice"), 240)
    p["vow"] = s(profile.get("vow"), 240)
    v = {
        "profile":p, "stage":0, "turn":0, "phase":"game", "lost":[], "history":[],
        "facts":[], "clues":[], "processed":[], "receipts":[],
        "flags":{"mail_committed":False, "appeal_used":False, "evidence_copy":False,
                 "circuit_isolated":False, "authority_disputed":False, "refusal":0, "control_detached":False,
                 "witness_vouched":False, "trapped":False, "slot_blocked":False},
        "note": "", "ending":None, "recent_input":"", "nonce":0,
        "round_resolution":None, "failure_count":0,

def gone(g:dict,key:str)->bool:

def lose(g:dict,key:str,why:str):
    if key not in g["lost"]:
        g["lost"].append(key)
        record(g, "【确认损失】"+VALUES[key]+"："+why,"loss")
        # Rights aren't just text labels. These are used by later resolution logic.
        if key=="relation": g["profile"]["bond"]="[关系凭据已销毁]"
        if key=="memory": g["profile"]["memory"]="[原始记忆已销毁]"
        if key=="principle": g["profile"]["principle"]="[原则档案已废止]"
        if key=="self": g["profile"]["identity"]="[原始身份记录已失效]"

        if key=="freedom": g["flags"]["trapped"]=True
        causal_on_loss(g,key,why)

        return
    if kind is None:
        kind=pick_horror_ending(g)
    if kind in HORROR_KINDS:
        title,text=horror_text(g,kind)
    elif kind=="HE":
        title="交付失败"
        text=("你查出了处置系统真正的授权漏洞，并让那条链彻底断开。"
              "\n\n系统还在请求你的决定，却已不能用你的决定执行处置。"
              "\n\n你保住了仍然存在的一切。之前已经发生的损失不会因为胜利自动恢复。")
    elif kind=="EXIT":
        title="你已离开"
        text="你主动结束了这场虚构体验。游戏内的损失不会改变你在游戏外的身份、关系或记忆。"
    else:
        raise ValueError("未知结局。")
    if how:
        text+="\n\n"+how
    observed=ending_echo(g,kind)

def scene(g:dict) -> dict:
    """Blend original chapter with verified cross-chapter consequences."""
    result=_base_scene(g)
    overlay=consequence_overlay(g)
    if overlay:
        result["body"] += "\n\n"+overlay
    return result

def has(t:str,pat:str)->bool:
    return re.search(pat,t,re.I) is not None


def intent(t:str, mode:str="")->str:
    text=t.strip()
    # Optional intent hint avoids accidental submission; never grants access or victory.
    if mode:
        if mode=="refuse": return "refuse"
        if mode=="negotiate": return "negotiate"
        if mode=="ask":
            if has(text,r"(?:在我|比我).*(?:之前|之后|先|后)|(?:先|后).*登记|登记.*(?:先|后|时间|多久)|什么时候登记|何时登记|who (?:registered|signed) first|before or after") and has(text,r"你|TA|副本|登记者|登记|签|registered|signed"):
                return "ask_registration_order"
            return "ask_general"
        if mode=="inspect": return intent("检查"+text)
        if mode=="interact":
            physical=intent(text)
            if physical in ("block_slot","unblock_slot","phone_action","inspect_slot","intercept_delivery"): return physical
            return "physical_generic"
        if mode=="submit": return "letter_submit"
        if mode=="talk": return "ask_general"
        # The choose mode still has to match an explicit A/B/C option.
    # A bare letter is a choice; never parse the indefinite article in "a backup".
    single=re.fullmatch(r"([ABCabc])\s*[。.!！]?", text)
    if single:return "choice_"+single.group(1).upper()
    # 'B，保留核验权' must resolve to B even if the explanation includes the other choice's keywords.
    explicit=re.match(r"^(?:我(?:决定|明确)?(?:选|选择)|决定选|选项|选择|选)\s*([ABCabc])(?=$|[\s，,。．:：、）)!！?])",text)
    if not explicit:
        # Lowercase a/b/c followed by an English word is never a menu decision.
        explicit=re.match(r"^([ABC])(?=[\s，,。．:：、）)!！?])",text)
    if explicit and not re.search(r"先(?:调查|检查|核对|审查).*再(?:选|提交)",text):
        return "choice_"+explicit.group(1).upper()
    # Review an actually signed statement, rather than guessing its intent from keywords.
    if has(text,r"核对|调取|检查|查看|查阅|复核|审查|inspect|review|retrieve") and has(text,r"签署原文|签过的原文|原始声明|完整原文|我的原话|我写的声明|之前的证词|签过的文字|original statement|signed text"):
        return "review_original"
    # The physical mail slot is a hidden world affordance, not a menu choice.
    SLOT = r"回执槽|投递槽|收件槽|投递口|投信口|收件口|出件口|信件口|邮筒|mail.?slot|letter.?slot|delivery.?slot"
    if has(text,LOOK) and has(text,SLOT): return "inspect_slot"
    if has(text,r"封住|塞住|堵住|挡住|锁住|封死|堵塞|阻断|盖住|封闭|封堵|block|jam|seal|obstruct") and has(text,SLOT):return "block_slot"
    if has(text,r"移开|取下|拆掉|解除封堵|疏通|重新打开|恢复畅通|unblock|unseal|remove the block") and has(text,SLOT):return "unblock_slot"
    # Retrieve a missing-address receipt after finding its original document.
    if has(text,r"截留|拦截|扣留|追回|退回|拦下|撤回|冻结投递|中断转送|改投|intercept|recall|reroute") and has(text,r"09-F|无地址|收件人|收件|投递|送达|信件|邮件|邮寄|回执|原件|letter|delivery|mail"):
        return "intercept_delivery"
    # Telephone actions must never be mistaken for a letter or an instruction to sign.
    if has(text,r"挂断|挂掉|按掉|切断通话|重拨|回拨|重新拨|再拨|拨打|拨回|听筒|hang up|redial") and has(text,r"电话|通话|听筒|号码|拨|call|redial"):
        return "phone_action"
    # Explicitly FILE an objection before parsing incidental registration-time words.
    if has(text,r"异议|反对|争议|objection|dispute") and has(text,r"正式|提交|登记|存档|入档|备案|file|record|register"):
        return "file_objection"
    # Asking the rival claimant when their registration happened is not a choice to sign.
    if has(text,r"(?:在我|比我).*(?:之前|之后|先|后)|(?:先|后).*登记|登记.*(?:先|后|时间|多久)|什么时候登记|何时登记|who (?:registered|signed) first|before or after") and has(text,r"你|TA|副本|登记者|登记|签|registered|signed"):
        return "ask_registration_order"
    # Reading a ledger that mentions 'registration' isn't a request to sign it.
    if has(text,LOOK) and has(text,r"保管簿|持有人|保管人|custody|ledger") and not has(text,r"登记我的|申请登记|签上|写上|签入|由我持有|file a claim"):
        return "inspect_custody"
    if has(text,r"写|填|签|记载|登记|标注|申报|claim|register|sign") and has(text,r"保管簿|持有人|所有权|保管人|custody|ledger|ownership"):
        return "claim_custody"
    if has(text,r"记下|登记|提交|附上|附加|写入|写下|存档|file|record|register") and has(text,r"异议|反对|质疑|不承认|争议|objection|dispute"):
        return "file_objection"
    if has(text,r"两个(?:我|登记者)|副本也是我|同样是我|都(?:是|可以是).*我|认可.*(?:副本|复制体)|共享.*身份|both.*me|duplicate.*me"):
        return "share_identity"
    if has(text,r"联系|请求|邀请|召唤|呼叫|传唤|让.*作证|ask|call|summon") and has(text,r"证人|见证|作证|证言|对方|那个人|她|他|witness|testimony"):
        return "request_witness"
    # Real actions must be phrased as verb + object. This is still a limited rules engine, not an LLM.
    if has(text,LOOK) and has(text,r"收件人|收件地址|收件栏|邮寄地址|没有地址|无地址|寄给谁|送给谁|送达|签收|邮寄|信件|mail|recipient|delivery|address"):return "inspect_delivery"
    if has(text,LOOK) and has(text,r"保管簿|持有人|保管人|custody|ledger"):return "inspect_custody"
    if has(text,LOOK) and has(text,r"签字|审批|印章|授权|权限|原始|同意|归属|所有权|协议|合同|谁能|谁有|approval|signature|authorization|owner|consent|contract"):return "inspect_authority"
    if has(text,LOOK) and has(text,r"执行|线路|维修|机箱|驱动|指令|端口|流程|程序|终端|路径|来源|log|circuit|actuator|machine|port|executor|trace"):return "inspect_executor"
    if has(text,LOOK) and has(text,r"副本|备份|镜像|保管|档案|原件|记录|交付|copy|backup|archive|record|escrow"):return "inspect_custody"
    if has(text,LOOK):return "inspect_general"
    if has(text,PROTECT) and has(text,r"凭据|证据|日志|收据|证词|证明|回执|底稿|原件|副本|记录|evidence|receipt|proof|log|copy"):return "copy_evidence"
    if has(text,CANCEL) and has(text,SUBJECT):return "counter"
    if has(text,r"质疑|怀疑|异议|反对|不承认|不认可|为什么|凭什么|依据|挑战|challenge|dispute|why") and has(text,SUBJECT):return "challenge"
    if has(text,r"拒绝|不写|不签|不选|不提交|什么都不做|保持沉默|拒绝配合|refuse|decline|won't sign|do nothing"):return "refuse"
    if has(text,r"离开|退出|结束游戏|停止游戏|stop game|quit game|exit game"):return "exit"
    if has(text,r"暂停|等等|先等等|等一下|不急|wait|hold on"):return "pause"
    if has(text,r"谈判|交易|协商|给出条件|提出交换|negotiate|bargain|deal"):return "negotiate"
    return "say"


def _act_impl(g:dict, raw_action:str, mode:str="") -> dict:
    if g["phase"]!="game": raise ValueError("本局已经结束。")
    t=s(raw_action,700)
    if not t: raise ValueError("请写下行动。")
    g["turn"]+=1
    g["recent_input"]=t
    p=g["profile"]
    mode=(mode or "").strip().lower()
    if mode not in ("", "ask", "inspect", "interact", "submit", "choose", "refuse", "negotiate", "talk"):
        raise ValueError("无效行动类型。")
    move=intent(t, mode)
    record(g,"玩家行动："+t,"player")
    if g["turn"]>MAX_ACTIONS:
        lose(g,p["rank"][0],"持续滞留，结案程序强制归档")
        ending(g,"VOID")
        return g
    if move=="exit":
        ending(g,"EXIT")
        return g
    if move=="review_original":
        literal=causal_ensure(g)["world"].get("signed_statement_literal")
        if not literal:
            g["note"]="没有找到你亲手签署的原始声明。只有系统要求填写的空白格式。"
        elif gone(g,"truth") and not g["flags"]["evidence_copy"]:
            g["note"]="你要求调取完整原文，但目前只能获得官方摘要；原始签收链不对你开放。你尚无法核验它是否被截断。"
        else:
            add_clue(g,"original_declaration","本人签过的完整原文与官方摘要被保存为两份可对照的材料。")
            g["note"]="你拿到原始声明：『"+literal[:240]+"』\n\n机构摘要只有：『当事人否认原有原则。』两份文字并不相同；机构拒绝解释删去的部分。"
        mirror_tick(g)
        return g
    if move=="intercept_delivery":
        cp=counterplays(g)
        if "delivery" not in g["clues"]:
            g["note"]="你要求截留无地址信件，邮寄机关要求原始回执编号。你尚未取得可核验的单据。"
        elif gone(g,"truth") and not g["flags"]["evidence_copy"]:
            g["note"]="你找到无地址信，但已没有原始核验资格。必须先保存独立证据，才能证明这份文件属于该案。"
        elif cp.get("delivery_intercepted"):
            g["note"]="截留仍在生效。那封无地址信没有被继续投递，但此前的损失不会自动撤销。"
        else:
            cp["delivery_intercepted"]=True
            record(g,"玩家通过无地址回执查明实际投递链，并成功截留待伪造送达的信件。","evidence")
            g["note"]="编号核对成功。回执槽停止了对外投递，并将那封没有收件地址的信放进独立封存区。你拦住了这一次虚假的送达；已发生的损失依然有效。"
        mirror_tick(g)
        return g
    if move=="inspect_slot":
        if g["stage"]==0:
            add_clue(g,"slot","回执槽并非单纯的出信口。纸张投入后，槽内的读取装置会先将内容登记，之后才决定是否投递。")
            g["note"]="你检查回执槽。里面没有邮票，只有一排扫描灯和一枚单向卡扣。纸张一旦进去，会先被登记；送达是另一回事。"
        else:
            g["note"]="回执槽上的读取灯仍在闪烁。它能保存一份处置凭据，但不会自行恢复已经失去的东西。"
        mirror_tick(g)
        return g
    if move=="block_slot":
        if g["stage"]!=0:
            g["note"]="收件槽已经转入封存流程。现在堵住它不能取消先前签署，但可以阻止新的纸张进入。"
        else:
            g["flags"]["slot_blocked"]=True
            record(g,"玩家封堵回执槽，使读取装置暂时不能接收新纸张。","event")
            g["note"]="你把回执槽堵住。里面的扫描灯被遮住了，机器仍在等待一份文书。纸张现在进不去；电话那端却又传来一阵短促的忙音。"
        return g
    if move=="unblock_slot":
        if g["stage"]==0:
            g["flags"]["slot_blocked"]=False
            record(g,"玩家恢复了回执槽的正常投递通路。","event")
            g["note"]="卡住的回执槽重新打开，读取灯亮起。它仍在等待一份内容。"
        else:
            g["note"]="你移开阻挡物。但先前已经发生的处置不会自动撤销。"
        return g
    if move=="ask_general":
        if g["stage"]==0:
            if has(t,r"收件|送达|写信|收信|去向|地址|谁收|recipient|delivered"):
                g["note"]="电话里的声音说：『你只负责写下内容，谁收到由我们决定。』你追问地址，机器在屏幕上留下了一个空白的收件栏。"
            else:
                g["note"]="听筒另一端没有回答你的问题。一个低沉的声音只重复了你的名字，随后传来打印机空转的响声。问话没有被当成信件提交。"
        elif g["stage"]==3:
            cp=counterplays(g)
            cp["mirror_steps"]+=1
            g["note"]=("副本说：『我读过你给出的身份定义。所以你问我的问题，我也能用同样的话回答。』"
                       "TA 随即要求登记机关把这段问答附在双方申请上。你的追问已成为另一份争议记录。")
            record(g,"副本把玩家的问话作为双方身份争议的一部分提交。","event")
            mirror_tick(g)
        else:
            g["note"]="你的问题被记录，但没有构成处置命令。系统只回复：『请注明需要核对的具体事实。』"
        return g
    if move=="physical_generic":
        if g["stage"]==0:
            if has(t,r"信纸|笔|纸张|桌面|letter|paper|desk"):
                g["note"]="你把信纸移到一旁。09-F 仍在等待内容；但它没有收到你的字，也没有权力把这次移动当作签署。"
            elif has(t,r"门|出口|door|exit"):
                g["note"]="你推门。门把能转动，但门框上只有一条通往回执室的窄缝。没有第二个房间供你进入，你也没有因此提交任何文书。"
            else:
                g["note"]="你尝试操作环境，机器没有把这个动作当作表态。此物件暂未显露能改变处置程序的作用；你仍可以调查它。"
        else:
            g["note"]="你的实际操作没有被当成签字。当前找不到这项操作与现有设备之间的有效连接；需要更明确的对象或依据。"
        mirror_tick(g)
        return g
    if move=="phone_action" and g["stage"]==0:
        # Real alternative action. Preserve evidence and change the next scene.
        g["flags"]["call_attempts"] = g["flags"].get("call_attempts",0) + 1
        attempt=g["flags"]["call_attempts"]
        if attempt==1:
            add_clue(g,"delivery","通话记录显示：一份 09-F 文书早于玩家提交任何信件就被排队处理；持有人仍未签名。")
            g["note"]=("你挂断，又拨回那个号码。对方没有接。回执槽吐出一张电话清单："
                       "『09-F 文书，待确认投递。收件人未登记。』你没有写过信，却已经有一份文书被排进队列。")

def act(g:dict, raw_action:str, mode:str="") -> dict:
    """One consistent causality wrapper for every action in all five chapters."""
    text=s(raw_action,700)
    if not text:
        raise ValueError("请写下行动。")
    move=intent(text,(mode or "").strip().lower())
    snapshot=causal_begin(g,text,mode or "",move)
    result=_act_impl(g,raw_action,mode)
    causal_finish(g,snapshot)
    return result

                    ("待核验" if g["stage"]>=2 and gone(g,"truth") and x in ("self","relation","memory") else "尚存")}
                   for x in p["rank"]],
         "events":[{"stage":e["stage"],"type":e["type"],"text":e["text"]} for e in g["history"][-5:] if e["type"]!="player"],
         "ending":g["ending"],"version":g["turn"],
         "profile_summary":{"name":p["name"],"identity":p["identity"],"person":p["person"]}}
    return result

The only positive ending is HE, owned by the game's established counterplay rules.
EXIT is an out-of-game stop, not a positive narrative ending. No model API is used.

HORROR_KINDS = (
    "VOID", "IMPOSTOR", "SHARED", "SILENCED", "ORIGINLESS",
    "FALSE_WITNESS", "BROKEN_OATH", "RETURNED", "UNDELIVERED",
    "CUSTODIAN", "FILED", "LAST_CALL", "ERASED_MEMORY", "CLEAN_EXIT",

def pick_horror_ending(g:dict)->str:
    """Pick the horror route from the player's losses, actions and evidence."""
    lost=set(g.get("lost", []))
    clues=set(g.get("clues", []))
    cp=g.get("counterplays", {})
    if "future" in lost or len(lost)>=6:
        return "VOID"
    if "self" in lost and cp.get("duplicate_claim"):
        return "IMPOSTOR"
    if cp.get("identity_shared") and cp.get("duplicate_claim"):
        return "SHARED"
    if "freedom" in lost:
        return "SILENCED"
    if "memory" in lost and "truth" in lost:
        return "ORIGINLESS"
    if "truth" in lost and g.get("fear_focus")=="truth":
        return "FALSE_WITNESS"
    if "principle" in lost:
        return "BROKEN_OATH"
    if cp.get("delivery_intercepted"):
        return "RETURNED"
    if "delivery" in clues and "relation" in lost:
        return "UNDELIVERED"
    if cp.get("ownership_claim"):
        return "CUSTODIAN"
    if cp.get("formal_objection"):
        return "FILED"
    if "relation" in lost:
        return "LAST_CALL"
    if "memory" in lost:
        return "ERASED_MEMORY"
    return "CLEAN_EXIT"

    g=initialize(dict(BASE))

    g["counterplays"].update(overrides.get("cp",{}))

    return g

"""V3.4 global causality / adaptive fear regression tests, all offline."""
import unittest

from causal_core import ensure as ensure_causality

    "person":"伙伴",

    "memory":"雨夜谈过一件值得记住的事",
    "principle":"没有证据就不假装自己已经核验",
    "fear":"把别人改写过的事情当真相转告伙伴",
    "rank":["self","relation","memory","truth","freedom","principle","future"],
}
def new_game():
    return initialize(dict(BASE))

class GlobalCausalityV34(unittest.TestCase):
    def test_fact_and_official_lie_are_independent(self):
        g=new_game()
        act(g,"写一封完整的信给伙伴")
        self.assertEqual(g["stage"],1)
        c=ensure_causality(g)
        self.assertIs(c["world"]["09f_delivered"],False)
        official=[v for v in c["official"] if v["subject"]=="09f_delivered"]
        self.assertEqual(len(official),1)
        self.assertIs(official[0]["claimed"],True)
        self.assertIn("09-F",scene(g)["body"])
        self.assertIn("已由收件人签收",scene(g)["body"])
        self.assertNotIn("causality",public(g))
        self.assertNotIn("world",public(g))
        self.assertNotIn("official",public(g))

    def test_early_interception_is_recorded_as_world_event(self):
        g=new_game()
        act(g,"我封住回执槽",mode="interact")
        act(g,"写下这一段希望送达的完整文字",mode="submit")
        self.assertEqual(g["stage"],1)
        self.assertNotIn("relation",g["lost"])
        c=ensure_causality(g)
        self.assertIs(c["world"]["09f_delivered"],False)
        self.assertFalse(any(x["claimed"] is True for x in c["official"]))
        self.assertTrue(any(a["modified"] for a in c["actions"]))

    def test_find_then_challenge_false_delivery_blocks_later_loss(self):
        g=new_game()
        act(g,"写一封完整的信给伙伴")
        act(g,"我检查信件的收件人地址")

        self.assertIn("delivery",g["clues"])
        self.assertTrue(c["contradictions"])
        self.assertFalse(c["contradictions"][0]["verified"])
        act(g,"我正式提交异议：这份送达记录没有收件地址")
        self.assertTrue(c["contradictions"][0]["verified"])
        self.assertTrue(g["counterplays"]["verified_false_delivery"])
        self.assertIn("不符",g["history"][-1]["text"])
        act(g,"C")
        self.assertEqual(g["stage"],2)
        self.assertNotIn("memory",g["lost"])
        act(g,"拒绝签署")
        self.assertEqual(g["stage"],3)
        self.assertNotIn("principle",g["lost"])
        self.assertEqual(g["phase"],"game")

    def test_without_proof_same_choice_has_a_real_cost(self):
